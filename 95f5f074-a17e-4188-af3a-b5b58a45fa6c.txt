function readProfile() {
  const ids=["age","gender","height","weight","goal","diet","region","likes","avoid","conditions","activity","workout","budget","mealCount","cooking"];
  return Object.fromEntries(ids.map(id => [id,$(id).value.trim()]));
}
function setProfile(p) {
  if (!p) return;
  Object.entries(p).forEach(([id,value]) => { if ($(id)) $(id).value=value; });
}
function storageGet(key, fallback) {
  try { return JSON.parse(localStorage.getItem(key)) ?? fallback; } catch { return fallback; }
}
function storageSet(key,value) {
  try { localStorage.setItem(key,JSON.stringify(value)); } catch {}
}
function calculateNutrition(p) {
  const bmr = p.gender==="female" ? 10*p.weight+6.25*p.height-5*p.age-161 : 10*p.weight+6.25*p.height-5*p.age+5;
  const activity = {low:1.2,moderate:1.45,high:1.65}[p.activity] || 1.45;
  const adjustment = {loss:-300,maintenance:0,gain:250,muscle:200}[p.goal] || 0;
  const calories = Math.max(1400,Math.round(bmr*activity+adjustment));
  const protein = Math.round(p.weight*({loss:1.2,maintenance:1.2,gain:1.4,muscle:1.6}[p.goal] || 1.2));
  const bmi = (p.weight/((p.height/100)**2)).toFixed(1);
  return {calories,protein,bmi};
}
function allowed(m,p) {
  const avoid = (p.avoid||"").toLowerCase().split(",").map(x=>x.trim()).filter(Boolean);
  if (avoid.some(x=>m.tags.includes(x))) return false;
  if (p.diet==="vegan" && (m.d==="Veg" || m.tags.includes("dairy"))) return false;
  if (p.diet==="veg" && m.d==="Non-Veg") return false;
  if (p.diet==="jain" && (m.tags.includes("onion") || m.tags.includes("garlic") || m.tags.includes("potato"))) return false;
  if (p.diet==="egg" && m.d==="Non-Veg") return false;
  return true;
}
function candidates(type,p) {
  const key = type==="breakfast" ? "breakfast" : type==="snack" ? "snack" : type;
  let list=meals.filter(m=>m.type===key&&allowed(m,p));
  const region = p.region==="north"?"north":p.region==="south"?"south":"generic";
  const regional=list.filter(m=>m.regions.includes(region));
  return regional.length ? regional : list;
}
function choose(type,p,offset=0) {
  const list=candidates(type,p);
  if (!list.length) return {n:"Fresh seasonal vegetables with grains",type,k:350,p:12,d:"Veg",tags:"vegetables grains",regions:"generic"};
  const liked=(p.likes||"").toLowerCase().split(",").map(x=>x.trim());
  const preferred=list.filter(m=>liked.some(x=>x&&m.tags.includes(x)));
  return (preferred.length?preferred:list)[offset%(preferred.length||list.length)];
}
function generateMealPlan(p) {
  const types=mealTypes[p.mealCount]||mealTypes[3];
  return days.map((day,di)=>({day,meals:types.map((name,i)=>{
    const type=i===0?"breakfast":name.toLowerCase().includes("snack")||name.includes("Workout")?"snack":name.toLowerCase();
    const m=choose(type,p,di+i);
    return {...m,name,time:times[p.mealCount][i],portion:m.type==="snack"?"1 bowl / serving":"1 practical serving"};
  })}));
}
function renderNutrition(n) {
  $("calories").textContent=n.calories;
  $("protein").textContent=n.protein+"g";
  $("bmi").textContent=n.bmi;
  const carbs=Math.round(n.calories*.5/4), fats=Math.round(n.calories*.25/9);
  $("macroTotal").textContent=`${carbs+fats+n.protein}g total`;
  $("macroBars").innerHTML=[
    ["Protein",n.protein,"g",Math.min(100,n.protein/1.8),"protein"],
    ["Carbohydrates",carbs,"g",50,"carbs"],["Fats",fats,"g",25,"fats"]
  ].map(x=>`<div class="macro-row"><div class="macro-label"><span>${x[0]}</span><strong>${x[1]}${x[2]} · ${Math.round(x[3])}%</strong></div><div class="track"><div class="fill ${x[4]}" style="width:${x[3]}%"></div></div></div>`).join("");
}

}
